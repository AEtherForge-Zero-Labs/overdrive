# Render benchmark (T036, US4; SC-003, SC-008). The program is ISO C, so
# CMake supplies what it cannot query portably: the compiler, build type,
# flags, target, and the configure host's processor, cores, and memory
# (research.md R47). They land in a generated header in the build tree.
function(ovd_bench_c_string out value)
    string(REPLACE "\\" "\\\\" value "${value}")
    string(REPLACE "\"" "\\\"" value "${value}")
    string(REPLACE "\n" " " value "${value}")
    string(STRIP "${value}" value)
    set("${out}" "${value}" PARENT_SCOPE)
endfunction()

cmake_host_system_information(RESULT bench_cpu QUERY PROCESSOR_DESCRIPTION)
cmake_host_system_information(RESULT bench_cores QUERY NUMBER_OF_PHYSICAL_CORES)
cmake_host_system_information(RESULT bench_memory QUERY TOTAL_PHYSICAL_MEMORY)
if(CMAKE_BUILD_TYPE)
    set(bench_build_type "${CMAKE_BUILD_TYPE}")
    string(TOUPPER "${CMAKE_BUILD_TYPE}" bench_config)
    set(bench_flags "${CMAKE_C_FLAGS} ${CMAKE_C_FLAGS_${bench_config}}")
else()
    set(bench_build_type "(none)")
    set(bench_flags "${CMAKE_C_FLAGS}")
endif()

ovd_bench_c_string(BENCH_COMPILER "${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}")
ovd_bench_c_string(BENCH_BUILD_TYPE "${bench_build_type}")
ovd_bench_c_string(BENCH_C_FLAGS "${bench_flags}")
ovd_bench_c_string(BENCH_TARGET "${CMAKE_SYSTEM_NAME} ${CMAKE_SYSTEM_PROCESSOR}")
ovd_bench_c_string(BENCH_HOST_SYSTEM "${CMAKE_HOST_SYSTEM_NAME} ${CMAKE_HOST_SYSTEM_VERSION}")
ovd_bench_c_string(BENCH_HOST_CPU "${bench_cpu}")
ovd_bench_c_string(BENCH_HOST_CORES "${bench_cores}")
ovd_bench_c_string(BENCH_HOST_MEMORY_MIB "${bench_memory}")

file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/bench_info.h" CONTENT [=[
/* Generated by bench/CMakeLists.txt at configure time. Do not edit. */
#ifndef OVERDRIVE_BENCH_INFO_H
#define OVERDRIVE_BENCH_INFO_H
#define BENCH_COMPILER "@BENCH_COMPILER@"
#define BENCH_BUILD_TYPE "@BENCH_BUILD_TYPE@"
#define BENCH_C_FLAGS "@BENCH_C_FLAGS@"
#define BENCH_TARGET "@BENCH_TARGET@"
#define BENCH_HOST_SYSTEM "@BENCH_HOST_SYSTEM@"
#define BENCH_HOST_CPU "@BENCH_HOST_CPU@"
#define BENCH_HOST_CORES "@BENCH_HOST_CORES@"
#define BENCH_HOST_MEMORY_MIB "@BENCH_HOST_MEMORY_MIB@"
#endif
]=] @ONLY)

add_executable(render_benchmark render_benchmark.c)
target_link_libraries(render_benchmark PRIVATE overdrive::overdrive)
target_include_directories(render_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
